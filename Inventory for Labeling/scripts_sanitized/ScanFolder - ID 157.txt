If [ $platform = -2 ]
	Set Variable [ $temp[1] ; Right ( $folderPath ; Length ( $folderPath ) - 1 ) ]
	Set Variable [ $outCSV[1] ; Right ( $outCSV ; Length ( $outCSV ) - 1 ) ]
	
	Set Variable [ $folderWin[1] ; Substitute ( $temp ; "/" ; "\\" ) ]
	
	Set Variable [ $quoted[1] ; "\"" & $folderWin & "\"" ]
	Set Variable [ $ps[1] ; "$DefaultMasks = @('*.3g2','*.3gp','*.aac','*.aif','*.aiff','*.amv','*.asf','*.avi','*.f4v','*.flac','*.flv','*.gifv','*.m2ts','*.m4a','*.m4v','*.mkv','*.mng','*.mov','*.mp3','*.mp4','*.mpg','*.mpeg','*.mts','*.mxf','*.nsv','*.ogg','*.rm','*.rmvb','*.roq','*.svi','*.ts','*.viv','*.vob','*.vtt','*.wav','*.webm','*.wma','*.wmv','*.yuv') " & Char ( 13 ) & Char ( 10 ) &
					
					"$AllowedExtensions = $DefaultMasks | ForEach-Object { $_ -replace '^\*\.', '' }" & Char ( 13 ) & Char ( 10 ) &
					
					"Get-ChildItem -LiteralPath " & $quoted & " -Recurse -File | " &

					"Select-Object FullName,Length | " &

	Set Variable [ $psscript[1] ; Get ( temporaryPath ) & "list.ps1" ]
	Set Variable [ $psscriptbat[1] ; Get ( temporaryPath ) & "list.bat" ]

	Create Data File [ "$psscript" ; Create folders: OFF ]
	Create Data File [ "$psscriptbat" ; Create folders: OFF ]

	Set Variable [ $i[1] ; 1 ]
	Loop [ Flush: Always ]
		Exit Loop If [ $i > $count ]
		Set Variable [ $line[1] ; GetValue ( $csvParsed; $i ) ]
		If [ Not IsEmpty ( $line ) ]
			Set Variable [ $fullPath[1] ; GetValue ( Substitute ( $line; "," ; "¶" ) ; 1) ]
