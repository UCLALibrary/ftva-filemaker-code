# exports Metadata from MetadataExport Layout in batches
Show All Records
Set Variable [ $totalRecords[1] ; Get(foundCount) ]
Set Variable [ $batchSize[1] ; 50000 ]
Set Variable [ $start[1] ; 1 ]
Set Variable [ $iteration[1] ; 1 ]

Loop [ Flush: Always ]
	Exit Loop If [ $start > $totalRecords ]
	# create found set for the current batch
	Go to Record/Request/Page [ First ]
	Omit Multiple Records [ With dialog: OFF ; $start - 1 ]
	Show Omitted Only
	Omit Multiple Records [ With dialog: OFF ; Get(foundCount) - $batchSize ]
	Pause/Resume Script [ Indefinitely ]
	# export found manually
	
	# update variables
	Set Variable [ $start[1] ; $start + $batchSize ]
	Set Variable [ $iteration[1] ; $iteration + 1 ]
	Show All Records
End Loop
